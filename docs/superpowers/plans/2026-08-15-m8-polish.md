import { idOffsetAngle } from '../src/sim/behaviors';
import { tick } from '../src/sim/Sim';
import { createWorld, spawnCreature, type WorldState } from '../src/sim/state';

describe('idOffsetAngle', () => {
  it('is deterministic and spreads ids around the circle', () => {
    expect(idOffsetAngle(7)).toBe(idOffsetAngle(7));
    const angles = [1, 2, 3, 4, 5, 6, 7, 8].map(idOffsetAngle);
    for (const a of angles) {
      expect(a).toBeGreaterThanOrEqual(0);
      expect(a).toBeLessThan(Math.PI * 2);
    }

```ts
/** Deterministic per-creature angle (same trick as voice.ts's roll): id-hash → [0, 2π). */
export function idOffsetAngle(id: number): number {
  let h = Math.imul(id, 0x85ebca6b) >>> 0;
  h ^= h >>> 15;
  h = Math.imul(h, 0x735a2d97) >>> 0;
  h ^= h >>> 13;
  return ((h >>> 0) / 4294967296) * Math.PI * 2;
}
```

In the socialize/court case, replace the "move toward partner" target with the ring point: `const ring = { x: partner.pos.x + Math.cos(idOffsetAngle(c.id)) * SOCIAL_RANGE * 0.45, y: partner.pos.y + Math.sin(idOffsetAngle(c.id)) * SOCIAL_RANGE * 0.45 };` — move toward `ring` while `dist(partner) > SOCIAL_RANGE * 0.7`, unchanged otherwise (social relief branch identical). In family.ts nesting, add the per-parent ring offset to the gather target. NO new rng draws anywhere.

### Task 3: Ambient effects — grass sway, water shimmer, dappled light, fireflies

**Files:** Create src/render/effects/Ambient.ts · Modify src/render/Renderer.ts, src/render/terrain/ValleyPainter.ts (export grass tuft positions if needed)

**Interfaces:** `class AmbientEffects { constructor(world: Container, cosmeticSeed: number); build(renderer: PixiRenderer): void; update(dtMs: number, clock: Clock, view: {x,y,zoom}): void }` — owns four sub-effects; Renderer calls `update` each frame after grading.

Implementation notes (all cosmetic RNG = seeded sfc32 from rng.ts with a COSMETIC seed, or Math.random at init only — never per-frame allocation):
- **Grass sway:** at build, scatter ~140 small grass-tuft Graphics (3 blade strokes each, 2 greens) across the meadow (reject pond/forest-core/grove-core via inEllipse). Per frame: `tuft.skew.x = Math.sin(tMs / periodMs + phase) * 0.06` with per-tuft period 2400–3600ms and phase from init RNG. Group them in one Container; skip updates entirely when `view.zoom < 0.35` (invisible at world view anyway) — cheap win.
- **Water shimmer:** two 256×256 soft-noise RenderTextures baked at init (draw ~200 low-alpha white ellipses on transparent), added as two TilingSprites masked to the pond ellipse (a Graphics mask), blendMode 'add', alpha 0.05/0.04, tilePosition scrolling in opposite directions ~6px/s. Night: alpha × (0.4 + 0.6 × light).
- **Dappled light:** 3 additive-blend soft radial blobs (baked circle texture, alpha 0.05, scale ~4–6) drifting on slow Lissajous paths over the meadow; visible only when light > 0.5 (fade with light).
- **Fireflies:** ONE ParticleContainer with ~40 4px warm dots (baked circle texture, tint 0xfff2b0). Each has an anchor point near the forest edge/meadow flowers + drifts on per-particle sine orbits; alpha = (1 − light) × twinkle(sin). Entirely hidden (`container.visible = light < 0.35`) by day. Fireflies visit memorials: bias 8 of the anchors to the 8 most recent memorial positions if any exist (re-read anchors from state on sync, not per frame).
